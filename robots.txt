# robots.txt для «Мастера Грузии» (mastera.ge)

# Сайт полностью открыт для индексации.
# Служебные страницы (корневой language-router, /<lang>/fb/, 404)
# закрыты метатегом noindex в самой разметке, а не здесь:
# robots.txt запретил бы их СКАНИРОВАНИЕ, и тогда краулер не увидел бы
# noindex — страница могла бы попасть в выдачу без описания.
User-agent: *
Allow: /

# Рекламные и реферальные метки не меняют содержимое страницы: сайт их вообще
# не читает (в script.js нет ни одного обращения к query-параметрам), поэтому
# /ru/?utm_source=ads и /ru/ — это одна и та же страница. Без этой директивы
# Яндекс считает их разными URL и тратит на дубли краулинговый бюджет.
# Google то же самое понимает по самоссылочному canonical, который стоит
# на всех 108 страницах, — для него директива не нужна и безвредна.
# ⚠️ Параметр ?v= здесь НЕ указан намеренно: он висит на CSS и JS, где служит
# версионированием кэша (см. bump-version.py), и склеивать его нельзя.
Clean-param: utm_source&utm_medium&utm_campaign&utm_term&utm_content&utm_referrer
Clean-param: gclid&gbraid&wbraid&yclid&fbclid&ysclid&_openstat

# Карта сайта — основной источник URL для краулеров.
Sitemap: https://mastera.ge/sitemap.xml
